13/**
14 * Drives one deploy-files content copy through five phases -- setup -> copy -> core -> finalize ->
15 * done -- doing at most one bounded unit of work per advance() call, so a poll loop (the REST advance
16 * route or the WP-cron fallback) can finish an arbitrarily large deploy without any single request
17 * running long.
18 *
19 * The three bash bookends run through Staging's existing command path (token handshake, env, escaping):
20 *   - setup    -> `deploy_content_setup`    (sweeps stale -new/-old; nothing else)
21 *   - core     -> `deploy_content_core`     (WordPress core download + php-ai-client, deliberately
22 *                                            after the copy so a long copy never leaves production
23 *                                            running new core against old content)
24 *   - finalize -> `deploy_content_finalize` (atomic promote of the finished -new tree over production)
25 *
26 * Between them, CopyEngine builds the `-new` hardlink-delta tree in bursts. Phase and progress are
27 * persisted under nfd-private so a crashed/closed request resumes instead of restarting, and every
28 * transition is mirrored into Staging's deploy-result file so the legacy `/staging/deploy` GET status
29 * poller and a recorded failure both keep working.

218    /**
219     * Do at most one bounded unit of work (one bash verb OR one engine burst) and return progress.
220     *
221     * The whole call runs under a non-blocking advisory lock, so a second caller -- a duplicate REST
222     * poll, the cron fallback, or a second browser tab -- can never run a phase transition concurrently
223     * (e.g. a setup that `rm -rf`s `-new` while another call's copy burst is filling it). A caller that
224     * cannot take the lock just reports current progress.
225     *
226     * @param bool $restart When true, discard any prior job (engine cursor/manifest + phase state) and
227     *                      start fresh. The UI passes this on the first advance of a new deploy so a
228     *                      previous run's terminal 'done'/'error' state does not short-circuit it.
229     *
230     * @return array { status: running|success|error, progress: array, phase: string, message?: string }
231     */
232    public function advance( $restart = false ) {
233        $dir = $this->jobDir();
234        if ( ! is_dir( $dir ) && ! wp_mkdir_p( $dir ) ) {
235            return array(
236                'status'   => 'error',
237                'phase'    => self::PHASE_ERROR,
238                'progress' => array(),
239                'message'  => __( 'Could not create the content copy job directory.', 'wp-module-staging' ),
240            );
241        }
242
243        $lock = fopen( $this->controllerLockPath(), 'cb' ); // phpcs:ignore WordPress.WP.AlternativeFunctions.file_system_operations_fopen
244        if ( false === $lock ) {
245            return array(
246                'status'   => 'error',
247                'phase'    => self::PHASE_ERROR,
248                'progress' => array(),
249                'message'  => __( 'Could not open the content copy lock.', 'wp-module-staging' ),
250            );
251        }
252        if ( ! flock( $lock, LOCK_EX | LOCK_NB ) ) {
253            fclose( $lock );
254            return $this->status(); // Another advance() is in flight; report without mutating.
255        }
256
257        try {
258            if ( $restart ) {
259                $this->resetLocked();
260            } elseif ( $this->jobExists() && $this->isStale() ) {
261                // A non-restart advance for a job that has not progressed in the stale window: it was
262                // abandoned (a tab asleep through a laptop suspend, say). Resuming it would run the copy
263                // against a tree another operation (clone/destroy) may have since mutated. Refuse and mark
264                // it failed; the client must start a fresh deploy (restart=true), which reseeds cleanly.
265                $stale          = $this->loadState( false );
266                $stale['phase'] = self::PHASE_ERROR;
267                $stale['message'] = __( 'The deploy was abandoned and timed out. Start it again.', 'wp-module-staging' );
268                $this->saveState( $stale );
269                return $this->envelope( 'error', $stale );
270            }
271
272            $state = $this->loadState( true );
273
274            switch ( $state['phase'] ) {
275                case self::PHASE_SETUP:
276                    return $this->runSetup( $state );
277                case self::PHASE_COPY:
278                    return $this->runCopyBurst( $state );
279                case self::PHASE_CORE:
280                    return $this->runCore( $state );
281                case self::PHASE_FINALIZE:
282                    return $this->runFinalize( $state );
283                case self::PHASE_ERROR:
284                    return $this->envelope( 'error', $state );
285                case self::PHASE_DONE:
286                default:
287                    return $this->envelope( 'success', $state );
288            }
289        } finally {
290            flock( $lock, LOCK_UN );
291            fclose( $lock );
292        }
293    }

295    /**
296     * Discard any prior job so a new deploy starts clean. Clears the engine's cursor/manifest/lock and
297     * the phase-state file together, so the next setup sweeps stale `-new`/`-old` and the engine reseeds
298     * rather than resuming a previous run's cursor against a swept tree. Call only under the lock.
299     *
300     * @return void
301     */
302    private function resetLocked() {
303        /*
304         * Release BEFORE the state file goes, because the state file holds the only record of the
305         * token. Deleting it first stranded the lock permanently: no later call could name the token,
306         * so nothing could ever release it and the site waited out the full TTL. That was reachable
307         * from an ordinary retry -- the UI resends `restart` after a gateway timeout, and a reload or
308         * a second tab does the same -- so the common case of "the first poll took too long" cost the
309         * user half an hour of refusals.
310         *
311         * A restart is also exactly when this is safe: the caller is abandoning the old job, so there
312         * is nothing of ours left that the lock protects. releaseIfOurs still declines a lock whose
313         * recorded holder is running, or cannot be shown not to be -- the one case where something of
314         * ours may still be writing, such as an orphaned setup part-way through its sweep.
315         */
316        if ( is_file( $this->statePath() ) ) {
317            $old_token = $this->jobToken( $this->loadState( false ) );
318            $this->lock()->releaseIfOurs( $old_token );
319            $this->withdrawCompatChannel( $old_token );
320        }
321
322        $this->engine()->cleanup();
323        if ( is_file( $this->statePath() ) ) {
324            wp_delete_file( $this->statePath() );
325        }

328    /**
329     * Withdraw this job's publication from the cross-version compat channel (lib/.staging's
330     * LEGACY_LOCK_NAME transient) so a restart or retry is not blocked by a token a failed job left
331     * behind.
332     *
333     * A stepped deploy's setup announces `active:<ts>:<token>` on this transient, so a peer running an
334     * older copy of the script (which cannot see the lock FILE) still sees the mutex held. If the job
335     * then fails while holding the lock, nothing withdraws that announcement: the failing verb's
336     * release path leaves the held lock in place, and the transient ages out only at LOCK_TTL. A fresh
337     * deploy carries a new token, so lock_check's legacy_lock_held sees a foreign, still-fresh holder
338     * and refuses every verb, blocking the retry the user just asked for for up to half an hour.
339     * Clearing our own prior publication on restart closes that window.
340     *
341     * Cleared by DB option, not delete_transient(): lib/.staging sets this transient with the object
342     * cache drop-in skipped, so it is always database backed, and legacy_lock_held reads it the same
343     * way. A delete_transient() from a request with a persistent object cache would clear only the
344     * cache copy and leave the row the script actually reads. Guarded by token so a concurrent
345     * operation's publication is never touched.
346     *
347     * @param string $token This job's lock token (the one being discarded).
348     *
349     * @return void
350     */
351    private function withdrawCompatChannel( $token ) {
352        if ( '' === $token ) {
353            return;
354        }
355
356        $value = get_option( '_transient_' . self::COMPAT_CHANNEL_TRANSIENT );
357        if ( ! is_string( $value ) ) {
358            return;
359        }
360
361        // active:<ts>:<token>
362        $parts = explode( ':', $value );
363        if ( count( $parts ) >= 3 && 'active' === $parts[0] && hash_equals( (string) $parts[2], $token ) ) {
364            delete_option( '_transient_' . self::COMPAT_CHANNEL_TRANSIENT );
365            delete_option( '_transient_timeout_' . self::COMPAT_CHANNEL_TRANSIENT );
366        }
367    }

427    /**
428     * Phase 1: run the bash setup bookend, then move to the copy phase.
429     *
430     * @param array $state Current state.
431     *
432     * @return array
433     */
434    private function runSetup( $state ) {
435        // Start the engine from a clean slate every time setup runs. A support engineer who deletes the
436        // phase-state file (but not the engine's cursor) would otherwise fall back to the setup phase
437        // while a STALE cursor still points mid-tree; the copy phase would then resume from it and
438        // promote only the tail of the tree over production. cleanup() removes the cursor/manifest so the
439        // engine reseeds, matching the stale -new/-old sweep the bash setup bookend does on the files.
440        $this->engine()->cleanup();
441
442        $this->writeRunning( $state );
443
444        $response = $this->callVerb( 'deploy_content_setup', $state );
445        if ( $this->verbFailed( $response ) ) {
446            // Setup is the verb most likely to outlive its request: its stale-`-new` sweep is one
447            // unlink per file. So it is also the one whose orphan the next poll runs into, and that
448            // refusal must not be read as a broken deploy.
449            $running = $this->ownStepStillRunning( $state );
450            if ( null !== $running ) {
451                return $running;
452            }
453
454            return $this->fail( $state, $this->verbMessage( $response, __( 'Content deploy setup failed.', 'wp-module-staging' ) ) );
455        }
456
457        $state['phase'] = self::PHASE_COPY;
458        $this->saveState( $state );
459        $this->writeRunning( $state );
460
461        return $this->envelope( 'running', $state );
462    }
463
464    /**
465     * Confirm this job still holds the staging mutex, and build the failure envelope if it does not.
466     *
467     * Every phase that is about to act on the trees has to ask, not just the copy. lib/.staging's
468     * lock_check cannot answer it for us: presented with no lock file at all it simply acquires one,
469     * so a verb could not tell "we have held this all along" from "we lost it, and whatever took it
470     * has since finished and released". The difference matters. The failure that motivates this: the
471     * last copy burst moves the job to the core phase, the poll driver then stalls for longer than
472     * the lock TTL but less than the job's stale window (a suspended laptop, a starved cron), a clone
473     * takes the lock, rewrites staging from production, finishes and releases -- and the core and
474     * finalize phases would then promote a `-new` tree built from the pre-clone staging content over
475     * production, because the clone only ever touched staging and left the `-new` tree standing.
476     *
477     * Either operation answers it, because both only succeed on a lock that exists and still carries
478     * this job's token -- false while a thief holds it and false after a thief has released it.
479     *
480     * WHICH ONE MATTERS. $refresh also WRITES, and the value it writes records pid 0, meaning "no
481     * single process owns this right now". Only the copy phase may say that: it is the one phase with
482     * nothing of the job's running between calls, and the refresh is also what keeps the lock from
483     * ageing out across a long copy. The phases that invoke a bash verb must NOT write, because the
484     * pid already in the lock may belong to a verb of this same job that is still running -- a
485     * finalize whose PHP request was killed by the FPM timeout while the bash process carried on
486     * swapping directories. That pid is the only thing standing between the next poll and a SECOND
487     * concurrent finalize: lock_check refuses to reclaim a lock whose recorded process is alive, and
488     * zeroing it here would tell it there is nobody to wait for, mid-swap. So core and finalize check
489     * read-only and leave the pid alone; the verb's own lock_check refreshes the timestamp a moment
490     * later anyway.
491     *
492     * @param array $state   Current state.
493     * @param array $failure Set to the failure envelope when this returns false.
494     * @param bool  $refresh Whether to also push the lock's timestamp forward (copy phase only).
495     *
496     * @return bool
497     */
498    private function holdsLockOrFail( $state, &$failure, $refresh ) {
499        $token = $this->jobToken( $state );
500        $held  = $refresh ? $this->lock()->refresh( $token ) : $this->lock()->holds( $token );
501
502        if ( $held ) {
503            return true;
504        }
505
506        $failure = $this->fail(
507            $state,
508            __( 'The deploy lost its lock to another staging operation and was stopped before changing production. Start the deploy again.', 'wp-module-staging' )
509        );
510
511        return false;
512    }
513
514    /**
515     * Phase 2: run one CopyEngine burst. Still reports running when the copy completes, because the
516     * core-download and finalize bookends have not run yet, so the poll driver keeps calling advance().
517     *
518     * @param array $state Current state.
519     *
520     * @return array
521     */
522    private function runCopyBurst( $state ) {
523        // Every burst re-asserts the mutex, which is also what keeps it alive: nothing of ours runs
524        // between bursts, so without a refresh here the lock setup took would age past its TTL
525        // mid-copy and lib/.staging would hand the mutex to the next operation that asked -- which
526        // would then mutate the very trees this job is enumerating and writing into.
527        if ( ! $this->holdsLockOrFail( $state, $failure, true ) ) {
528            return $failure;
529        }
530
531        $result            = $this->engine()->advance();
532        $state['progress'] = isset( $result['progress'] ) ? $result['progress'] : array();
533
534        if ( 'error' === $result['status'] ) {
535            return $this->fail( $state, isset( $result['message'] ) ? $result['message'] : __( 'Copying content files failed.', 'wp-module-staging' ) );
536        }
537
538        if ( 'success' === $result['status'] ) {
539            $state['phase'] = self::PHASE_CORE;
540        }
541
542        $this->saveState( $state );
543        $this->writeRunning( $state );
544
545        return $this->envelope( 'running', $state );
546    }
547
548    /**
549     * Phase 2.5: run the bash core-download bookend (WordPress core + php-ai-client), then move to
550     * finalize. Kept distinct from finalize so finalize is only the instant swap -- see
551     * deploy_content_core in lib/.staging. Still reports running so the poll driver continues.
552     *
553     * @param array $state Current state.
554     *
555     * @return array
556     */
557    private function runCore( $state ) {
558        // Read-only: this phase invokes a bash verb, so it must not overwrite the pid in the
559        // lock. See holdsLockOrFail.
560        if ( ! $this->holdsLockOrFail( $state, $failure, false ) ) {
561            return $failure;
562        }
563
564        $response = $this->callVerb( 'deploy_content_core', $state );
565        if ( $this->verbFailed( $response ) ) {
566            $running = $this->ownStepStillRunning( $state );
567            if ( null !== $running ) {
568                return $running;
569            }
570
571            return $this->fail( $state, $this->verbMessage( $response, __( 'Preparing the WordPress core for deploy failed.', 'wp-module-staging' ) ) );
572        }
573
574        $state['phase'] = self::PHASE_FINALIZE;
575        $this->saveState( $state );
576        $this->writeRunning( $state );
577
578        return $this->envelope( 'running', $state );
579    }
580
581    /**
582     * Phase 3: run the bash finalize bookend, clean up the engine, mark done.
583     *
584     * @param array $state Current state.
585     *
586     * @return array
587     */
588    private function runFinalize( $state ) {
589        /*
590         * A finalize this job already ran may have SUCCEEDED without this phase ever being told.
591         *
592         * deploy_content_finalize swaps, releases the mutex, and only THEN deletes the `-old` asides,
593         * which on a normal deploy is the only step whose cost scales with the site's file count (the
594         * sweep in deploy_content_setup is the other, and only after an abandoned attempt). The point of that
595         * ordering is that the request blocking on it is free to die: by then production is live and
596         * correct. But the request dying is also what stops saveState(PHASE_DONE) below from running,
597         * so the job is still recorded as `finalize`, and the next poll re-enters here.
598         *
599         * Without this check, that poll found the lock gone (the verb released it) and reported "the
600         * deploy lost its lock and was stopped BEFORE changing production" -- on a deploy that had
601         * completed. That is the exact symptom the release-then-delete ordering was built to remove,
602         * reintroduced on the PHP side.
603         */
604        if ( $this->swapAlreadyCompleted( $state ) ) {
605            return $this->finishLocked( $state );
606        }
607
608        // Read-only: this phase invokes a bash verb, so it must not overwrite the pid in the
609        // lock. See holdsLockOrFail.
610        if ( ! $this->holdsLockOrFail( $state, $failure, false ) ) {
611            return $failure;
612        }
613
614        /*
615         * Recorded BEFORE the verb runs, and durably, because the process that learns the answer may
616         * not survive to report it. It is what lets the check above tell "a finalize of ours ran and
617         * the trees are gone because it promoted them" from "the copy phase produced nothing to
618         * promote in the first place". Both look identical on disk.
619         */
620        if ( empty( $state['finalize_attempted'] ) ) {
621            $state['finalize_attempted'] = true;
622            $this->saveState( $state );
623        }
624
625        $response = $this->callVerb( 'deploy_content_finalize', $state );
626        if ( $this->verbFailed( $response ) ) {
627            // An earlier finalize of this same job may have got further than the one we just tried.
628            if ( $this->swapAlreadyCompleted( $state ) ) {
629                return $this->finishLocked( $state );
630            }
631
632            $running = $this->ownStepStillRunning( $state );
633            if ( null !== $running ) {
634                return $running;
635            }
636
637            return $this->fail( $state, $this->verbMessage( $response, __( 'Finalizing the content deploy failed.', 'wp-module-staging' ) ) );
638        }
639
640        return $this->finishLocked( $state );
641    }
642
643    /**
644     * The `running` envelope to return when a verb was refused because an earlier verb of THIS job is
645     * still working, or null when that is not what happened.
646     *
647     * lock_check will not reclaim a lock whose recorded process is alive, and it reports that with the
648     * same customer-facing message as every other refusal, so the message cannot be used to tell them
649     * apart. When the live holder is one of our own verbs -- a bookend that outlived the request that
650     * started it and is still running -- the job is progressing, not broken, and the poll driver has
651     * to be told to come back rather than shown a failure.
652     *
653     * Deliberately does NOT save the phase state. advance()'s stale window is measured from the last
654     * state write, so leaving it alone is what bounds this: a verb that hangs for ever ends as a
655     * failed job instead of polling for ever.
656     *
657     * @param array $state Current state.
658     *
659     * @return array|null
660     */
661    private function ownStepStillRunning( array $state ) {
662        if ( ! $this->lock()->heldByOwnLiveProcess( $this->jobToken( $state ) ) ) {
663            return null;
664        }
665
666        $this->writeRunning( $state );
667
668        return $this->envelope( 'running', $state );
669    }
670
671    /**
672     * Whether the swap this job was running has already finished, leaving nothing to promote.
673     *
674     * Only ever true once a finalize has actually been invoked (see `finalize_attempted`): a job whose
675     * staging source had no content directories at all would otherwise have no `-new` trees from the
676     * start and would be reported as a completed swap without the verb ever running, which would also
677     * skip releasing the mutex.
678     *
679     * "Nothing to promote" is the same test bash itself uses to decide a directory is already done
680     * (see finalize_new_content_dirs), and the states it could be confused with do not arise: a verb
681     * killed mid-swap leaves the later directories' `-new` in place AND leaves its lock behind, and a
682     * verb that failed and rolled back restores `-new` from each directory it had promoted.
683     *
684     * @param array $state Current state.
685     *
686     * @return bool
687     */
688    private function swapAlreadyCompleted( array $state ) {
689        if ( empty( $state['finalize_attempted'] ) ) {
690            return false;
691        }
692
693        $config = $this->engineConfig();
694        foreach ( CopyEngine::DEFAULT_ROOTS as $dir ) {
695            if ( is_dir( $config['dest_root'] . '/' . $dir . $config['dest_suffix'] ) ) {
696                return false;
697            }
698        }
699
700        return true;
701    }
702
703    /**
704     * Record a completed deploy. Shared by the normal path and by the resumption check above, so a
705     * deploy finished by an orphaned verb is reported exactly like one finished in place.
706     *
707     * @param array $state Current state.
708     *
709     * @return array
710     */
711    private function finishLocked( array $state ) {
712        $this->engine()->cleanup();
713
714        /*
