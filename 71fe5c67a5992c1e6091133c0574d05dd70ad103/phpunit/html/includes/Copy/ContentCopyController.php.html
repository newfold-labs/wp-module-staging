212     * poll, the cron fallback, or a second browser tab -- can never run a phase transition concurrently
213     * (e.g. a setup that `rm -rf`s `-new` while another call's copy burst is filling it). A caller that
214     * cannot take the lock just reports current progress.

217     *                      start fresh. The UI passes this on the first advance of a new deploy so a
218     *                      previous run's terminal 'done'/'error' state does not short-circuit it.

227                'phase'    => self::PHASE_ERROR,
228                'progress' => array(),
229                'message'  => __( 'Could not create the content copy job directory.', 'wp-module-staging' ),
230            );
231        }
232
233        $lock = fopen( $this->controllerLockPath(), 'cb' ); // phpcs:ignore WordPress.WP.AlternativeFunctions.file_system_operations_fopen
234        if ( false === $lock ) {
235            return array(
236                'status'   => 'error',
237                'phase'    => self::PHASE_ERROR,
238                'progress' => array(),
239                'message'  => __( 'Could not open the content copy lock.', 'wp-module-staging' ),
240            );

287     * the phase-state file together, so the next setup sweeps stale `-new`/`-old` and the engine reseeds

353        if ( ! $active ) {
354            return false;
355        }
356
357        // An abandoned job (crashed/closed before reaching a terminal phase) would otherwise block every
358        // later staging operation forever, since nothing ever flips its phase to done/error. Treat a job
359        // that has not advanced within the stale window as dead, not in flight.
360        return ! $this->isStale();
361    }

363    /**
364     * Whether a job's phase-state file has not been touched within the stale window. advance() rewrites
365     * it on every burst (writeRunning/saveState), so an older mtime means no progress for that long.
366     * Only meaningful when a job exists.
367     *
368     * @return bool
369     */
370    private function isStale() {
371        $mtime = @filemtime( $this->statePath() );
372        return false !== $mtime && $mtime < ( time() - self::STALE_AFTER_SECONDS );
373    }

375    /**
376     * Phase 1: run the bash setup bookend, then move to the copy phase.
377     *
378     * @param array $state Current state.
379     *
380     * @return array
381     */
382    private function runSetup( $state ) {
383        // Refuse before touching anything if another staging operation already holds the site-wide
384        // mutex. lib/.staging's lock_check would refuse the setup verb anyway (it is the authority on
385        // the lock, and applies a stronger liveness test than this one), so this is only a fast path:
386        // it keeps a doomed deploy from clearing the engine's cursor and spending an exec, and gives a
387        // message that names the real reason.
388        if ( $this->lock()->heldByOther( $this->jobToken( $state ) ) ) {
389            return $this->fail(
390                $state,
391                __( 'Another staging operation is already running. Wait for it to finish, then start the deploy again.', 'wp-module-staging' )
392            );
393        }
394
395        // Start the engine from a clean slate every time setup runs. A support engineer who deletes the
396        // phase-state file (but not the engine's cursor) would otherwise fall back to the setup phase
397        // while a STALE cursor still points mid-tree; the copy phase would then resume from it and
398        // promote only the tail of the tree over production. cleanup() removes the cursor/manifest so the
399        // engine reseeds, matching the stale -new/-old sweep the bash setup bookend does on the files.
400        $this->engine()->cleanup();
401
402        $this->writeRunning( $state );
403
404        $response = $this->callVerb( 'deploy_content_setup', $state );
405        if ( $this->verbFailed( $response ) ) {
406            return $this->fail( $state, $this->verbMessage( $response, __( 'Content deploy setup failed.', 'wp-module-staging' ) ) );
407        }
408
409        $state['phase'] = self::PHASE_COPY;
410        $this->saveState( $state );
411        $this->writeRunning( $state );
412
413        return $this->envelope( 'running', $state );
414    }

416    /**
417     * Confirm this job still holds the staging mutex, and build the failure envelope if it does not.
418     *
419     * Every phase that is about to act on the trees has to ask, not just the copy. lib/.staging's
420     * lock_check cannot answer it for us: presented with no lock file at all it simply acquires one,
421     * so a verb could not tell "we have held this all along" from "we lost it, and whatever took it
422     * has since finished and released". The difference matters. The failure that motivates this: the
423     * last copy burst moves the job to the core phase, the poll driver then stalls for longer than
424     * the lock TTL but less than the job's stale window (a suspended laptop, a starved cron), a clone
425     * takes the lock, rewrites staging from production, finishes and releases -- and the core and
426     * finalize phases would then promote a `-new` tree built from the pre-clone staging content over
427     * production, because the clone only ever touched staging and left the `-new` tree standing.
428     *
429     * Either operation answers it, because both only succeed on a lock that exists and still carries
430     * this job's token -- false while a thief holds it and false after a thief has released it.
431     *
432     * WHICH ONE MATTERS. $refresh also WRITES, and the value it writes records pid 0, meaning "no
433     * single process owns this right now". Only the copy phase may say that: it is the one phase with
434     * nothing of the job's running between calls, and the refresh is also what keeps the lock from
435     * ageing out across a long copy. The phases that invoke a bash verb must NOT write, because the
436     * pid already in the lock may belong to a verb of this same job that is still running -- a
437     * finalize whose PHP request was killed by the FPM timeout while the bash process carried on
438     * swapping directories. That pid is the only thing standing between the next poll and a SECOND
439     * concurrent finalize: lock_check refuses to reclaim a lock whose recorded process is alive, and
440     * zeroing it here would tell it there is nobody to wait for, mid-swap. So core and finalize check
441     * read-only and leave the pid alone; the verb's own lock_check refreshes the timestamp a moment
442     * later anyway.
443     *
444     * @param array $state   Current state.
445     * @param array $failure Set to the failure envelope when this returns false.
446     * @param bool  $refresh Whether to also push the lock's timestamp forward (copy phase only).
447     *
448     * @return bool
449     */
450    private function holdsLockOrFail( $state, &$failure, $refresh ) {
451        $token = $this->jobToken( $state );
452        $held  = $refresh ? $this->lock()->refresh( $token ) : $this->lock()->holds( $token );
453
454        if ( $held ) {
455            return true;
456        }
457
458        $failure = $this->fail(
459            $state,
460            __( 'The deploy lost its lock to another staging operation and was stopped before changing production. Start the deploy again.', 'wp-module-staging' )
461        );
462
463        return false;
464    }
465
466    /**
467     * Phase 2: run one CopyEngine burst. Still reports running when the copy completes, because the
468     * core-download and finalize bookends have not run yet, so the poll driver keeps calling advance().
469     *
470     * @param array $state Current state.
471     *
472     * @return array
473     */
474    private function runCopyBurst( $state ) {
475        // Every burst re-asserts the mutex, which is also what keeps it alive: nothing of ours is
476        // running between bursts (setup exits as soon as it has swept the stale trees, and core is not
477        // due until the copy finishes, possibly hours later on a large site), so without a refresh here
478        // the lock setup took would age past its TTL mid-copy and lib/.staging would hand the mutex to
479        // the next clone/create/destroy/deploy that asked for it -- which would then mutate the very
480        // trees this job is enumerating and writing into.
481        if ( ! $this->holdsLockOrFail( $state, $failure, true ) ) {
482            return $failure;
483        }
484
485        $result            = $this->engine()->advance();
486        $state['progress'] = isset( $result['progress'] ) ? $result['progress'] : array();
487
488        if ( 'error' === $result['status'] ) {
489            return $this->fail( $state, isset( $result['message'] ) ? $result['message'] : __( 'Copying content files failed.', 'wp-module-staging' ) );
490        }
491
492        if ( 'success' === $result['status'] ) {
493            $state['phase'] = self::PHASE_CORE;
494        }
495
496        $this->saveState( $state );
497        $this->writeRunning( $state );
498
499        return $this->envelope( 'running', $state );
500    }
501
502    /**
503     * Phase 2.5: run the bash core-download bookend (WordPress core + php-ai-client), then move to
504     * finalize. Kept distinct from finalize so finalize is only the instant swap -- see
505     * deploy_content_core in lib/.staging. Still reports running so the poll driver continues.
506     *
507     * @param array $state Current state.
508     *
509     * @return array
510     */
511    private function runCore( $state ) {
512        // Read-only: this phase runs a bash verb, so it must not overwrite the pid in the lock. See
513        // holdsLockOrFail.
514        if ( ! $this->holdsLockOrFail( $state, $failure, false ) ) {
515            return $failure;
516        }
517
518        $response = $this->callVerb( 'deploy_content_core', $state );
519        if ( $this->verbFailed( $response ) ) {
520            return $this->fail( $state, $this->verbMessage( $response, __( 'Preparing the WordPress core for deploy failed.', 'wp-module-staging' ) ) );
521        }
522
523        $state['phase'] = self::PHASE_FINALIZE;
524        $this->saveState( $state );
525        $this->writeRunning( $state );
526
527        return $this->envelope( 'running', $state );
528    }
529
530    /**
531     * Phase 3: run the bash finalize bookend, clean up the engine, mark done.
532     *
533     * @param array $state Current state.
534     *
535     * @return array
536     */
537    private function runFinalize( $state ) {
538        // Read-only, and here it matters most: an orphaned finalize of this same job may still be
539        // swapping directories, and the pid in the lock is what stops this poll starting a second one.
540        // See holdsLockOrFail.
541        if ( ! $this->holdsLockOrFail( $state, $failure, false ) ) {
542            return $failure;
543        }
544
545        $response = $this->callVerb( 'deploy_content_finalize', $state );
546        if ( $this->verbFailed( $response ) ) {
547            return $this->fail( $state, $this->verbMessage( $response, __( 'Finalizing the content deploy failed.', 'wp-module-staging' ) ) );
548        }
549
550        $this->engine()->cleanup();
551
552        $state['phase'] = self::PHASE_DONE;
553        $this->saveState( $state );
554        $this->writeSuccess( $state );
555
556        return $this->envelope( 'success', $state );
557    }

559    /**
560     * Invoke a bash verb through the configured runner, handing it this job's staging-lock token.
561     *
562     * The token is what makes the three verbs one operation as far as lib/.staging's lock_check is
563     * concerned: each presents it, recognizes the lock already carrying it as its own, and refreshes
564     * that lock instead of blocking on it or taking a second one.
565     *
566     * @param string $verb  Verb name.
567     * @param array  $state Current state (carries the lock token).
568     *
569     * @return array|\WP_Error
570     */
571    private function callVerb( $verb, array $state ) {
572        return call_user_func( $this->verb_runner, $verb, $this->jobToken( $state ) );
573    }

576     * Whether a verb response indicates failure.
577     *
578     * @param mixed $response Verb response.
579     *
580     * @return bool
581     */
582    private function verbFailed( $response ) {
583        if ( is_wp_error( $response ) ) {
584            return true;
585        }
586
587        return is_array( $response ) && isset( $response['status'] ) && 'error' === $response['status'];
588    }

591     * Extract a human message from a failed verb response.
592     *
593     * @param mixed  $response Verb response.
594     * @param string $default  Fallback message.
595     *
596     * @return string
597     */
598    private function verbMessage( $response, $default ) {
599        if ( is_wp_error( $response ) ) {
600            return $response->get_error_message();
601        }
602
603        if ( is_array( $response ) && ! empty( $response['message'] ) ) {
604            return (string) $response['message'];
605        }

610    /**
611     * Record a failure: mark the error phase and persist it to the result file.
612     *
613     * @param array  $state   Current state.
614     * @param string $message Failure message.
615     *
616     * @return array
617     */
618    private function fail( $state, $message ) {
619        $state['phase']   = self::PHASE_ERROR;
620        $state['message'] = $message;
621        $this->saveState( $state );
622
623        // This job is over, so stop holding the staging mutex: a failure during the copy phase would
624        // otherwise block every other staging operation until the lock aged out. Declines unless the
625        // lock is provably ours alone and nothing else could be mid-flight under it -- see
626        // StagingCommandLock::releaseIfOurs.
627        $this->lock()->releaseIfOurs( $this->jobToken( $state ) );
628
629        $this->staging->recordDeployResult(
630            array(
631                'status'     => 'error',
632                'command'    => self::DEPLOY_COMMAND,
633                'started_at' => $state['started_at'],
634                'message'    => $message,
635            )
636        );
637
638        return $this->envelope( 'error', $state, $message );
639    }

771    /**

775     * is the wrong tool for it.

783    /**
784     * Persist the phase state atomically (temp file + rename).
785     *
786     * @param array $state State to persist.
787     *
788     * @return void
789     */
790    private function saveState( array $state ) {
791        $dir = $this->jobDir();
792        if ( ! is_dir( $dir ) && ! wp_mkdir_p( $dir ) ) {
793            return;
794        }
795
