523     *
524     * @param string   $command          CLI command name.
525     * @param int|null $since_timestamp  Only consider log lines at or after this Unix time.
526     *

528     */
529    protected function getDeployStatusFromLog( $command, $since_timestamp = null ) {
530        $log_file = trailingslashit( $this->getProductionDir() ) . 'nfd-private/nfd-staging.log';
531        if ( ! file_exists( $log_file ) ) {

538        }

540        $lines = file( $log_file, FILE_IGNORE_NEW_LINES | FILE_SKIP_EMPTY_LINES ); // phpcs:ignore WordPress.WP.AlternativeFunctions.file_get_contents_file_get_contents

558                );
559            }
560            if ( false !== strpos( $line, '[SUCCESS]' ) && false !== strpos( $line, '[' . $success_step . ']' ) ) {
561                return array(

563                    'command' => $command,
564                    'message' => $this->getDeploySuccessMessage( $command ),

649     * Start deploy in a shutdown handler so the REST response can return immediately.

651     * @param string $script         Path to the staging shell script.
652     * @param string $command_name   CLI command name (e.g. deploy_db).

731    /**
732     * Run the staging shell script and parse its JSON stdout.
733     *
734     * @param string $script       Path to the staging shell script.
735     * @param string $command_name CLI command name (e.g. deploy_db), used to gate the compat_check exemptions below.
736     * @param string $command      Escaped CLI argument string.
737     *
738     * @return array|\WP_Error
739     */
740    protected function executeStagingScript( $script, $command_name, $command ) {
741        do_action( 'newfold_staging_command', $command ); // bh_staging_command
742
743        $output      = array();
744        $exit_status = 0;
745        exec( "{$script} {$command}", $output, $exit_status ); // phpcs:ignore
746
747        // The script wrote options with the cache bypassed, so put the two back in sync.
748        if ( 'compat_check' !== $command_name ) {
749            $this->resync_object_cache();
750        }
751
752        /*
753         * auth_action() consumes the token, but only on the paths that reach it. compat_check exits
754         * before authenticating, and the script can die earlier than that. Unlike the transient this
755         * replaced, an option does not expire on its own, so an uncollected token would sit in
756         * wp_options in plain text and travel into every database export.
757         */
758        delete_option( 'staging_auth_token' );
759
760        return $this->parseStagingScriptResponse( $output, $exit_status );
761    }

763    /**
764     * Decode JSON stdout from the staging shell script.
765     *
766     * @param array $output      Lines of stdout captured from the staging script.
767     * @param int   $exit_status Exit status of the staging script.
768     *
769     * @return array|\WP_Error
770     */
771    protected function parseStagingScriptResponse( $output, $exit_status = 0 ) {
772        $output = (array) $output;
773
774        /*
775         * Take the last line that actually parses as one of our responses rather than assuming the
776         * script's own output is the last thing on stdout. A drop-in, a PHP notice or any other
777         * stray write would otherwise be handed to json_decode() and turn a working run into the
778         * generic failure below. Every response the script emits carries a "status" key.
779         */
780        $response = null;
781        for ( $i = count( $output ) - 1; $i >= 0; $i-- ) {
782            $decoded = json_decode( trim( $output[ $i ] ), true );
783            if ( is_array( $decoded ) && isset( $decoded['status'] ) ) {
784                $response = $decoded;
785                break;
786            }
787        }
788
789        if ( null === $response ) {
790            $last_line  = end( $output );
791            $cloudflare = $last_line ? json_decode( trim( $last_line ), true ) : null;
792            if ( is_array( $cloudflare ) && ! empty( $cloudflare['cloudflare_error'] ) ) {
793                return new \WP_Error(
794                    'origin_timeout',
795                    __( 'The deployment is still running. Please wait and check again shortly.', 'wp-module-staging' ),
796                    array( 'status' => 524 )
797                );
798            }
799
800            return new \WP_Error( 'json_decode', __( 'Something gone wrong, please get in touch with our support.', 'wp-module-staging' ) );
801        }
802
803        /*
804         * Some commands print their success line and then keep working (deploy_db cleans up its
805         * backup and runs the exit trap afterwards). If the script died during that tail we would
806         * otherwise find the success line and report a run that did not finish as successful, so
807         * the exit status has the final say.
808         */
809        if ( 0 !== $exit_status && 'error' !== $response['status'] ) {
810            return new \WP_Error( 'error_response', __( 'The staging operation did not complete, please get in touch with our support.', 'wp-module-staging' ) );
811        }
812
813        /*
814         * Array access, not object access: json_decode() above is associative, so an object-style
815         * test could never match and error messages from the script would be silently returned to
816         * the caller as if they were successful responses.
817         */
818        if ( isset( $response['status'], $response['message'] ) && 'error' === $response['status'] ) {
819            return new \WP_Error( 'error_response', $response['message'] );
820        }
821
822        return $response;
823    }
